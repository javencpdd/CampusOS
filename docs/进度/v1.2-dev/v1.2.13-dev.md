```python
from pathlib import Path
import yaml
for name in ['ci_test.yml', 'deploy.yml']:
    workflow = yaml.safe_load((Path('.github/workflows') / name).read_text())
    checks = 0
    for job in workflow['jobs'].values():
        steps = job.get('steps', [])
        for i, step in enumerate(steps):
            if 'make contracts-check' not in step.get('run', ''):
                continue
            prior = steps[:i]
            assert any(x.get('uses', '').startswith('actions/setup-node@') for x in prior)
            assert any(x.get('uses', '').startswith('pnpm/action-setup@') for x in prior)
            assert any(x.get('run', '') == 'pnpm --dir sdk/typescript install --frozen-lockfile' for x in prior)
            checks += 1
    assert checks == 1
```
